cmake_minimum_required(VERSION 3.25)
set(K210_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}")
include(external/k210_sdk_import.cmake)
project(k210_sdk LANGUAGES C CXX ASM)
k210_sdk_init()
option(K210_BUILD_EXAMPLES "Build SDK examples" ON)
option(K210_BUILD_TESTS "Build SDK integration fixtures" OFF)
if(K210_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
if(K210_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
